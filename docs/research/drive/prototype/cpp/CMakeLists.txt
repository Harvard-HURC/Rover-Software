cmake_minimum_required(VERSION 3.20)
project(rover_drive_proto CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
add_library(RoverDrive SHARED rover_drive.cpp)
target_link_libraries(RoverDrive PRIVATE gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13)
target_compile_options(RoverDrive PRIVATE -Wall -Wextra)
